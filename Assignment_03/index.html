function connect() { var searchTerm = document.getElementById("searchBox").value; var url = `https://www.themealdb.com/api.php${searchTerm}`; fetch(url) .then(res => res.json()) .then(data => display(data)); } function display(data) { var oldContent = document.getElementById("displayArea"); oldContent.textContent = ""; // if nothing is found, the API sends meals: null if (data.meals == null) { oldContent.textContent = "No meals found. Try another name."; return; } // show only the first 5 results for (var i = 1; i <= 5 && i <= data.meals.length; i++) { var newDiv = document.createElement("div"); newDiv.classList.add("innerStyle"); newDiv.innerHTML = `
Meal Title: ${data.meals[i-1].strMeal}
Category: ${data.meals[i-1].strCategory}
Area: ${data.meals[i-1].strArea}`; oldContent.appendChild(newDiv); } }
